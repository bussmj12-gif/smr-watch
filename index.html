// pages/api/smr/save.js // 선택된 기사들을 카드로 다듬어 저장합니다. // 저장할 때마다 기존 카드는 "보관"(is_current=false) 처리하고, 새로 선택한 것만 "현재 브리핑"(is_current=true)이 됩니다. import { createClient } from '@supabase/supabase-js'; const supabase = createClient( process.env.NEXT_PUBLIC_SUPABASE_URL, process.env.SUPABASE_SERVICE_ROLE_KEY ); export default async function handler(req, res) { res.setHeader('Access-Control-Allow-Origin', 'https://bussmj12-gif.github.io'); res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') return res.status(200).end(); if (req.method !== 'POST') return res.status(405).json({ error: 'POST만 허용됩니다' }); const { selected } = req.body || {}; if (!Array.isArray(selected) || selected.length === 0) { return res.status(400).json({ error: '선택된 항목(selected)이 필요합니다' }); } try { const response = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': process.env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01', }, body: JSON.stringify({ model: 'claude-sonnet-4-5', max_tokens: 8192, system: '아래 기사 정보를 뉴스 카드 형식으로 다듬어라. 출력은 JSON 배열만, 다른 텍스트 없이.\n' + '[{"tag":"국내|해외|인증|투자|기술 중 하나","title":"간결하고 임팩트 있는 제목(한글)",' + '"summary":"카드에 보일 1~2문장 핵심 요약(한글)",' + '"detail":"신문 기사처럼 배경, 맥락, 의미, 향후 전망까지 포함한 5~8문장 상세 설명(한글). 단순 요약이 아니라 왜 중요한지, 관련 업계에 어떤 영향이 있는지까지 풀어서 서술한다.",' + '"source":"출처명","date":"YYYY-MM-DD","url":"원문 링크"}]', messages: [ { role: 'user', content: `다음 기사들을 카드로 다듬어줘:\n${JSON.stringify(selected)}` }, ], }), }); const data = await response.json(); const textBlock = (data.content || []).find((b) => b.type === 'text'); if (!textBlock) throw new Error('Claude 응답에 텍스트가 없습니다'); const cleaned = textBlock.text.replace(/```json|```/g, '').trim(); let cards; try { cards = JSON.parse(cleaned); } catch (e) { console.error('JSON 파싱 실패, 원문:', cleaned); throw new Error('카드 형식 변환에 실패했습니다 (응답이 잘렸을 수 있음)'); } cards = cards.map((c) => ({ ...c, is_current: true })); const { error: archiveError } = await supabase .from('smr_articles') .update({ is_current: false }) .eq('is_current', true); if (archiveError) throw archiveError; const { error: insertError } = await supabase.from('smr_articles').insert(cards); if (insertError) throw insertError; return res.status(200).json({ saved: cards.length }); } catch (err) { console.error(err); return res.status(500).json({ error: err.message || '저장 중 오류가 발생했습니다' }); } }
